# Запуск сервера:
#   node main.js -h 127.0.0.1 -p 3000 -c ./cache

# 1. GET перший (MISS)
curl -s -D - -o /tmp/cat-404.jpg http://127.0.0.1:3000/404

# 2. GET повторний (HIT)
curl -s -D - -o /tmp/cat-404b.jpg http://127.0.0.1:3000/404

# 3. Інші коди
curl -s -D - -o /tmp/cat-200.jpg http://127.0.0.1:3000/200
curl -s -D - -o /tmp/cat-418.jpg http://127.0.0.1:3000/418

# 4. GET неіснуючого коду
curl -s -D - http://127.0.0.1:3000/999

# 5. PUT власної картинки
curl -s -D - -X PUT --data-binary @/tmp/cat-404.jpg http://127.0.0.1:3000/250

# 6. GET після PUT
curl -s -D - -o /tmp/cat-250.jpg http://127.0.0.1:3000/250

# 7. DELETE
curl -s -D - -X DELETE http://127.0.0.1:3000/250

# 8. DELETE коли файлу немає
curl -s -D - -X DELETE http://127.0.0.1:3000/250

# 9. Недозволений метод
curl -s -D - -X POST http://127.0.0.1:3000/200

# 10. Некоректний шлях
curl -s -D - http://127.0.0.1:3000/abc
curl -s -D - http://127.0.0.1:3000/
